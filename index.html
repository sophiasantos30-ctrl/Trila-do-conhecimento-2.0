const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); // Variável para guardar quem está no servidor // Formato: { 'id_do_socket': { username: 'João', roomId: 'sala1' } } const usuariosConectados = {}; app.get('/', (req, res) => { res.send(` Sala em Tempo Real

Jogadores na Sala

Carregando...
    `); }); io.on('connection', (socket) => { socket.on('joinRoom', ({ roomId, username }) => { socket.join(roomId); // Salva o usuário na memória do servidor usuariosConectados[socket.id] = { id: socket.id, username, roomId }; // Filtra apenas os usuários que estão nesta sala específica const usuariosNaSala = Object.values(usuariosConectados).filter(u => u.roomId === roomId); // Envia a lista completa para TODOS na sala io.to(roomId).emit('updateUsersList', usuariosNaSala); }); // Remove o usuário da lista se ele fechar a aba ou cair a internet socket.on('disconnect', () => { const usuario = usuariosConectados[socket.id]; if (usuario) { const roomId = usuario.roomId; delete usuariosConectados[socket.id]; // Remove da memória // Pega a nova lista sem o usuário que saiu const usuariosRestantes = Object.values(usuariosConectados).filter(u => u.roomId === roomId); // Atualiza a tela de quem ficou na sala io.to(roomId).emit('updateUsersList', usuariosRestantes); } }); }); server.listen(3000, () => { console.log('Servidor rodando em http://localhost:3000'); });
